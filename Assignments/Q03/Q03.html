class Point {
    int x, y;
public:
    Point(int x, int y) {
        x = x;
        y = y;
    }
    friend ostream& operator<<(ostream& os, const Point& p);
};
ostream& operator<<(ostream& os, const Point& p) {
    return os << "(" << p.x << ", " << p.y << ")";
}
int main() {
    Point p(3, 4);
    cout << p;   // (0, 0) -- wrong!
}
