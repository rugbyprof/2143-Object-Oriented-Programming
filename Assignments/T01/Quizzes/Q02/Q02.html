Question Add an overloaded constructor for the Stack class. 
class Stack {
    int *array;
    int top;

    Stack() {
        array = new int[10];
        top = -1;
    }

public:
    void push(int val) { array[++top] = val; }
};

int main() {
    Stack S;
    Stack S1(100)
}
